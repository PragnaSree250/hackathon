def decode_image_bytes(img_bytes: bytes):
    try:
        nparr = np.frombuffer(img_bytes, np.uint8)
        img = cv2.imdecode(nparr, cv2.IMREAD_COLOR)
        if img is not None:
            return img
    except Exception:
        pass

    try:
        pil_img = Image.open(io.BytesIO(img_bytes)).convert("RGB")
        return cv2.cvtColor(np.array(pil_img), cv2.COLOR_RGB2BGR)
    except Exception:
        return None

def crop_banknote_region(img):
    """
    Isolates the main rectangular banknote contour if surrounded by a plain background.
    """
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    blurred = cv2.GaussianBlur(gray, (5, 5), 0)
    _, thresh = cv2.threshold(blurred, 200, 255, cv2.THRESH_BINARY_INV)

    contours, _ = cv2.findContours(thresh, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
    if contours:
        c = max(contours, key=cv2.contourArea)
        x, y, w, h = cv2.boundingRect(c)
        if w > img.shape[1] * 0.4 and h > img.shape[0] * 0.4:
            return img[y:y+h, x:x+w]
            
    return img

def analyze_security_features(img_bytes: bytes):
    raw_img = decode_image_bytes(img_bytes)

    if raw_img is None:
        return False, 0.0, ["FAILED: Unable to decode image file."]

    # Auto-crop bounding box if isolated
    img = crop_banknote_region(raw_img)
    
    height, width, _ = img.shape
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    hsv = cv2.cvtColor(img, cv2.COLOR_BGR-HSV if hasattr(cv2, 'COLOR_BGR-HSV') else cv2.COLOR_BGR2HSV)

    reasons_failed = []
    verifications = []

    # ==========================================
    # 1. Security Thread OVI Color Verification (Green/Cyan)
    # ==========================================
    # Vertical band around x = 52% to 62%
    thread_crop = hsv[:, int(width * 0.52):int(width * 0.62)]
    
    # Range for Security Thread Green & Metallic Cyan
    green_mask = cv2.inRange(thread_crop, np.array([35, 30, 30]), np.array([88, 255, 255]))
    green_ratio = np.count_nonzero(green_mask) / float(thread_crop.shape[0] * thread_crop.shape[1])

    if green_ratio > 0.008:
        verifications.append(f"PASSED: Security Thread OVI green/teal color presence confirmed ({round(green_ratio*100, 2)}%).")
    else:
        reasons_failed.append("FAILED: Security thread lacks expected Optically Variable Ink (OVI) color profile.")

    # ==========================================
    # 2. Serial Number Character Variance Check
    # ==========================================
    # Bottom-Right Serial Number region (x = 65% to 92%, y = 75% to 95%)
    serial_crop = gray[int(height * 0.75):int(height * 0.95), int(width * 0.65):int(width * 0.92)]
    serial_std = float(np.std(serial_crop)) if serial_crop.size > 0 else 0.0

    # Real varying digits (e.g. 0VK 211320) produce std dev > 25.0
    # Blank prints or low quality placeholders give std dev < 18.0
    if serial_std < 18.0:
        reasons_failed.append(f"FAILED: Serial number zone lacks digit definition (StdDev: {round(serial_std, 1)}).")
    else:
        verifications.append(f"PASSED: Valid serial number character variation detected (StdDev: {round(serial_std, 1)}).")

    # ==========================================
    # 3. Watermark Window Multi-Tone Contrast
    # ==========================================
    # Unprinted window (x = 68% to 83%, y = 20% to 75%)
    wm_crop = gray[int(height * 0.20):int(height * 0.75), int(width * 0.68):int(width * 0.83)]
    if wm_crop.size > 0:
        wm_std = float(np.std(wm_crop))
        wm_mean = float(np.mean(wm_crop))

        if wm_mean > 100.0 and 12.0 <= wm_std <= 65.0:
            verifications.append(f"PASSED: Watermark window portrait shading depth verified (StdDev: {round(wm_std, 1)}).")
        else:
            reasons_failed.append("FAILED: Watermark window lacks multi-tone depth or is obscured.")

    # ==========================================
    # Final Decision Calculation
    # ==========================================
    is_authentic = len(reasons_failed) == 0
    confidence = 96.5 if is_authentic else 15.0
    all_messages = verifications + reasons_failed

    return is_authentic, confidence, all_messages


@app.post("/api/scan-currency")
async def scan_currency(file: UploadFile = File(...)):
    contents = await file.read()
    if not contents:
        raise HTTPException(status_code=400, detail="Empty file uploaded.")

    current_time = time.strftime("%H:%M:%S")
    is_authentic, confidence, verifications = analyze_security_features(contents)

    if not is_authentic:
        return {
            "is_fake": True,
            "status": "🔴 COUNTERFEIT / SUSPECTED FAKE NOTE",
            "file_name": file.filename,
            "scan_timestamp": current_time,
            "denomination_detected": "₹500 (Suspected Fake Note)",
            "confidence_score": confidence,
            "verifications": verifications,
            "citizen_advisory": "ALERT: Note failed key optical security checks (Security thread or Watermark missing)."
        }

    return {
        "is_fake": False,
        "status": "🟢 AUTHENTICATED PASS",
        "file_name": file.filename,
        "scan_timestamp": current_time,
        "denomination_detected": "₹500 Banknote (Genuine)",
        "confidence_score": confidence,
        "verifications": verifications,
        "citizen_advisory": "SPECIMEN PASSED: Image passed key optical, thread, and structural checks."
    }
